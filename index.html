Google Sheets setup

Step 1 — Create your Google Sheet

Go to sheets.google.com, create a new sheet called SBS Jam Feedback, and add these column headers in row 1:

Timestamp  |  Jam  |  Role  |  Rating  |  What Worked  |  To Improve  |  Comment Good  |  Comment Improve

Step 2 — Add the Apps Script

In your sheet go to Extensions → Apps Script. Delete everything and paste this code, then click Save (💾):

function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var type = e.parameter.type || 'response'; if (type === 'response') { var sheet = ss.getSheetByName('Responses') || ss.getActiveSheet(); var d = JSON.parse(e.postData.contents); sheet.appendRow([ d.timestamp, d.jam, d.role, d.rating, d.whatWorked, d.toImprove, d.commentGood, d.commentImprove ]); } if (type === 'saveJam') { var js = ss.getSheetByName('Jams'); if (!js) { js = ss.insertSheet('Jams'); js.appendRow(['id','name','venue','type','schedule']); } var d = JSON.parse(e.postData.contents); js.appendRow([d.id, d.name, d.venue, d.type, d.schedule]); } if (type === 'deleteJam') { var js = ss.getSheetByName('Jams'); if (!js) return ContentService.createTextOutput('OK'); var d = JSON.parse(e.postData.contents); var rows = js.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][0] == d.id) { js.deleteRow(i + 1); break; } } } return ContentService.createTextOutput('OK'); } function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var type = e.parameter.type || 'jams'; if (type === 'jams') { var js = ss.getSheetByName('Jams'); if (!js) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); var rows = js.getDataRange().getValues(); var jams = rows.slice(1).map(function(r){ return {id:r[0],name:r[1],venue:r[2],type:r[3],schedule:r[4]}; }); return ContentService.createTextOutput(JSON.stringify(jams)).setMimeType(ContentService.MimeType.JSON); } if (type === 'responses') { var rs = ss.getSheetByName('Responses') || ss.getActiveSheet(); var rows = rs.getDataRange().getValues(); var resps = rows.slice(1).map(function(r){ return { timestamp:r[0], jam:r[1], role:r[2], rating:r[3], whatWorked:r[4], toImprove:r[5], commentGood:r[6], commentImprove:r[7] }; }); var jamFilter = e.parameter.jam; if (jamFilter && jamFilter !== 'all') resps = resps.filter(function(r){ return r.jam === jamFilter; }); return ContentService.createTextOutput(JSON.stringify(resps)).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput('OK'); }
